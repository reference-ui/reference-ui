## Longer snippet

Tall blocks start collapsed and reveal the rest on demand.

```tsx
import { Field, Input, Label } from '@reference-ui/react'

type EmailFieldProps = {
  value: string
  onChange: (value: string) => void
  error?: string
}

export function EmailField({ value, onChange, error }: EmailFieldProps) {
  const describedBy = error ? 'email-error' : undefined

  return (
    <Field invalid={Boolean(error)}>
      <Label htmlFor="email">Email address</Label>
      <Input
        id="email"
        type="email"
        value={value}
        aria-describedby={describedBy}
        onChange={event => onChange(event.currentTarget.value)}
      />
      {error ? <Field.Error id="email-error">{error}</Field.Error> : null}
    </Field>
  )
}
```
